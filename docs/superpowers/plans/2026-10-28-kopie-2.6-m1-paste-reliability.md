# Kopie 2.6 — M1 Paste Reliability Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stories S-001…S-004 of `docs/STORIES.md` — Secure Input gate, one-retry paste for hostile targets, graceful clipboard+toast fallback (never silent), and rebindable paste keys with conflict detection.

**Architecture:** Pure decision logic moves into `KopieCore` (unit-testable): a `PastePlan` state machine decides post/retry/fallback from three inputs. The app-target `PasteDirectService` becomes a thin executor that consults the plan, posts ⌘V toward the *target process* (`CGEvent.postToPid`) instead of blind session taps, and reports obstacles. `AppDelegate.firePaste()` becomes the retry orchestrator (it owns focus restore). Failure toasts move from a popover-only `CopiedToast` to a shared `ToastCenter` + floating `ToastPanel` (NSPanel, non-activating) so they survive popover close. Binding persistence lands in `SettingsStore`; conflict checking is pure `ShortcutConflict` code in KopieCore; `PopoverView` reads bindings instead of hardcoded ⌥/⌃.

**Tech Stack:** Swift 6 strict concurrency, SwiftUI + AppKit, Carbon `RegisterEventHotKey`, `CGEventSource`/`CGEvent`, SQLite-free — no new dependencies. KopieCore unit tests via Swift Testing (`Tests/KopieCoreTests/`).

```swift
// Sources/KopieCore/Paste/PastePlan.swift
import Foundation
import os

/// Why a direct paste did not happen. The clip is ALWAYS already staged on the
/// system pasteboard before a paste op starts, so every failure path leaves the
/// user holding the copy — the toast only has to explain what happened.
public enum PasteObstacle: String, Equatable, Sendable {
    case secureInput
    case axMissing
    case targetRefused

    /// Mid-op failure signals ordered most-user-hostile first.
    public static func detect(axGranted: Bool, secureInputActive: Bool, targetStillFront: Bool) -> PasteObstacle? {
        secureInputActive ? .secureInput : (!axGranted ? .axMissing : (!targetStillFront ? .targetRefused : nil))
    }
}

/// What to do with a direct-paste attempt right now.
public enum PasteOutcome: Equatable, Sendable {
    case post                         // stage is on the pasteboard; post ⌘V now
    case retry                        // exactly one retry left: refocus target, re-post
    case fallback(PasteObstacle)      // do NOT post; surface toast
}

/// Inputs the plan needs; pure value so it can cross actor boundaries.
public struct PasteInput: Equatable, Sendable {
    public var axGranted: Bool
    public var secureInputActive: Bool
    public var attemptsMade: Int   // synthetic ⌘V posts already sent for this op (0-based)

    public init(axGranted: Bool, secureInputActive: Bool, attemptsMade: Int) {
        self.axGranted = axGranted
        self.secureInputActive = secureInputActive
        self.attemptsMade = attemptsMade
    }
}

public enum PastePlan {
    /// Decision table (S-001/S-002/S-003 of docs/STORIES.md):
    /// 1. Secure Input active → never post, fallback (S-001).
    /// 2. Missing Accessibility → fallback + "Open Settings…" toast (S-003).
    /// 3. attempt 0 → post; attempt 1 → retry; anything ≥ 2 → target refused (S-002).
    public static func decide(_ input: PasteInput) -> PasteOutcome {
        if input.secureInputActive { return .fallback(.secureInput) }
        if !input.axGranted { return .fallback(.axMissing) }
        switch input.attemptsMade {
        case 0: return .post
        case 1: return .retry
        default: return .fallback(.targetRefused)
        }
    }
}

public enum PasteToastCopy {
    /// Each obstacle gets its own sentence; `targetName` renders the target app's
    /// name so the user can connect the failure to the app it tried to reach.
    public static func message(obstacle: PasteObstacle, targetName: String?) -> String {
        switch obstacle {
        case .secureInput:
            return "Kopie can't paste while a password field is active. The clip is on your clipboard."
        case .axMissing:
            return "Kopie needs Accessibility access to paste into \(targetName ?? "this app"). The clip is on your clipboard."
        case .targetRefused:
            return "Unable to paste into \(targetName ?? "the app"). The clip is on your clipboard."
        }
    }
}

public enum PasteLog {
    /// One log line per paste decision, so a user report ("paste didn't happen")
    /// can be answered without attaching a debugger ("logs captured").
    public static let logger = Logger(subsystem: "com.kopie.app", category: "paste")
}
```

```swift
import AppKit
import ApplicationServices
import KopieCore

/// Pastes the current clipboard contents into the frontmost application by
/// simulating ⌘V, closing the copy→paste loop in one keystroke (Maccy-style
/// ⌥↩ / ⌥-click). Requires the Accessibility (AXIsProcessTrusted) permission.
///
/// All *decisions* live in `KopieCore.PastePlan` (unit-tested); this type only
/// reads system state and posts the events.
@MainActor
enum PasteDirectService {

    /// True when Kopie may synthesize key events (Accessibility granted).
    static var isPermissionGranted: Bool {
        AXIsProcessTrusted()
    }

    /// True while Secure Input is active system-wide (password field focused).
    /// Synthetic key events posted now are silently swallowed by the system,
    /// so the paste plan aborts before posting (S-001).
    static var isSecureInputActive: Bool {
        CGEventSource.secureInputEnabled()
    }

    /// Opens the System Settings pane where the user can grant Accessibility.
    nonisolated static func requestPermission() {
        DispatchQueue.global().async {
            let options = ["AXTrustedCheckOptionPrompt": true] as CFDictionary
            _ = AXIsProcessTrustedWithOptions(options)
        }
    }

    /// Jumps straight to Privacy & Security → Accessibility, where the grant
    /// lives.
    static func openAccessibilitySettings() {
        let candidates = [
            "x-apple.systempreferences:com.apple.preference.security?Privacy_Accessibility",
            "x-apple.systempreferences:com.apple.preference.security",
        ]
        for candidate in candidates {
            if let url = URL(string: candidate), NSWorkspace.shared.open(url) { return }
        }
    }

    /// The plan-consulting executor. `postToPID` routes the ⌘V into the target
    /// app's own event queue instead of the session-wide tap — that queue-hop is
    /// what keeps Electron/terminal/VM targets from eating the keystroke (S-002).
    /// Returns the plan outcome; on `.fallback` nothing was posted and the caller
    /// surfaces the obstacle (the clip remains staged on the pasteboard).
    @discardableResult
    static func attempt(delayMs: UInt32 = 80,
                        attemptsMade: Int = 0,
                        postToPID: pid_t? = nil) -> PasteOutcome {
        let input = PasteInput(axGranted: isPermissionGranted,
                               secureInputActive: isSecureInputActive,
                               attemptsMade: attemptsMade)
        let decision = PastePlan.decide(input)
        PasteLog.logger.info("decision=\(decision-logString) attempts=\(attemptsMade) secure=\(isSecureInputActive) ax=\(isPermissionGranted)")
        switch decision {
        case .post, .retry:
            schedulePost(delayMs: delayMs, postToPID: postToPID)
        case .fallback:
            break
        }
        return decision
    }

    /// Legacy boolean wrapper retained until Task 3 rewires the last callers.
    @discardableResult
    static func paste(delayMs: UInt32 = 80) -> Bool {
        if case .fallback = attempt(delayMs: delayMs) { return false }
        return true
    }

    private static func schedulePost(delayMs: UInt32, postToPID: pid_t?) {
        let post = { postKeyEvents(postToPID: postToPID) }
        if delayMs > 0 {
            DispatchQueue.main.asyncAfter(deadline: .now() + .milliseconds(Int(delayMs))) {
                MainActor.assumeIsolated { post() }
            }
        } else {
            post()
        }
    }

    /// 0x09 = v. Posts down then ⌘V up with a 10ms gap, into `pid`'s queue when
    /// the target is another process, else the session-wide event tap.
    private static func postKeyEvents(postToPID: pid_t?) {
        let source = CGEventSource(stateID: .combinedSessionState)
        let down = CGEvent(keyboardEventSource: source, virtualKey: 0x09, keyDown: true)
        down?.flags = [.maskCommand]
        let up = CGEvent(keyboardEventSource: source, virtualKey: 0x09, keyDown: false)
        up?.flags = [.maskCommand]
        guard let down, let up else { return }
        if let pid = postToPID, pid != ProcessInfo.processInfo.processIdentifier {
            down.postToPid(pid)
            usleep(10_000)
            up.postToPid(pid)
        } else {
            down.post(tap: .cghidEventTap)
            usleep(10_000)
            up.post(tap: .cghidEventTap)
        }
    }
}

private extension PasteOutcome {
    /// os.Logger interpolation accepts a String; map the enum once here.
    var logString: String {
        switch self {
        case .post: return "post"
        case .retry: return "retry"
        case .fallback(let obstacle): return "fallback:\(obstacle.rawValue)"
        }
    }
}
```

```swift
    /// Runs the direct-paste op against the app that was frontmost before the
    /// popover opened: plan → post → watch → at most one retry → toast on
    /// fallback. Replaces the old "Paste Blocked" notification path (S-003).
    private func firePaste() {
        firePasteStage(attemptsMade: 0)
    }

    private func firePasteStage(attemptsMade: Int) {
        let targetName = prePopoverApp?.localizedName
        let outcome = PasteDirectService.attempt(delayMs: 0,
                                                 attemptsMade: attemptsMade,
                                                 postToPID: prePopoverApp?.processIdentifier)
        if case .fallback(let obstacle) = outcome {
            ToastCenter.shared.showPaste(obstacle: obstacle, targetName: targetName)
            if obstacle == .axMissing { requestAccessibilityOnce() }
            return
        }
        // 150ms after posting, check the signals the plan can't see by itself:
        // the target lost frontmost (tap went to the wrong queue), Secure Input
        // turned on, or the AX grant was pulled mid-flight.
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.15) { [weak self] in
            MainActor.assumeIsolated {
                guard let self else { return }
                let stillFront = pasteTargetIsFrontmost(self.prePopoverApp)
                if let obstacle = PasteObstacle.detect(axGranted: PasteDirectService.isPermissionGranted,
                                                       secureInputActive: PasteDirectService.isSecureInputActive,
                                                       targetStillFront: stillFront) {
                    if attemptsMade == 0 {
                        restoreFrontmostApp()
                        DispatchQueue.main.asyncAfter(deadline: .now() + 0.03) { [weak self] in
                            MainActor.assumeIsolated { self?.firePasteStage(attemptsMade: 1) }
                        }
                    } else {
                        ToastCenter.shared.showPaste(obstacle: obstacle, targetName: self.prePopoverApp?.localizedName)
                    }
                }
                // Clean post (no obstacle): nothing further to do.
            }
        }
    }

    /// True while the expected paste target still owns the frontmost seat.
    private func pasteTargetIsFrontmost(_ target: NSRunningApplication?) -> Bool {
        guard let target else { return true }   // nothing to hold Kopie back
        return NSWorkspace.shared.frontmostApplication?.processIdentifier == target.processIdentifier
    }
```

```swift
// Sources/Kopie/UI/ToastCenter.swift
import SwiftUI

/// One floating toast at a time, app-wide, that survives popover close. Every
/// paste-failure surface routes through here so the wording, dedupe, and
/// lifetime live in exactly one place.
@MainActor
final class ToastCenter: ObservableObject {
    static let shared = ToastCenter()

    struct Message: Identifiable, Equatable {
        let id = UUID()
        var text: String
        var warning: Bool
        var actionTitle: String?
        var action: (() -> Void)?
        var dismissTitle: String?
        var onDismiss: (() -> Void)?

        static func == (l: Message, r: Message) -> Bool { l.id == r.id }
    }

    @Published private(set) var current: Message?
    /// One toast per session per reason — no toast spam when several pastes
    /// fail the same way (S-003 states).
    private var deliveredReasons: Set<PasteObstacle> = []
    private var hideWork: DispatchWorkItem?

    /// The three paste obstacles (S-001/S-003). Dashboard of copy lives in
    /// `PasteToastCopy`; Secure Input toasts auto-fade in 6s, the others 8s.
    func showPaste(obstacle: PasteObstacle, targetName: String?) {
        guard !deliveredReasons.contains(obstacle) else { return }
        deliveredReasons.insert(obstacle)
        let seconds: Double = obstacle == .secureInput ? 6 : 8
        show(text: PasteToastCopy.message(obstacle: obstacle, targetName: targetName),
             warning: true,
             seconds: seconds,
             reason: obstacle,
             actionTitle: obstacle == .secureInput ? nil : "Open Settings…",
             action: obstacle == .secureInput ? nil : { PasteDirectService.openAccessibilitySettings() },
             dismissTitle: "Dismiss",
             onDismiss: { [weak self] in self?.dismiss() })
    }

    /// Generic entry used by the in-popover warning toasts after Task 5's
    /// wiring; `reason` (when given) participates in the once-per-session dedupe.
    func show(text: String,
              warning: Bool,
              seconds: Double,
              reason: PasteObstacle? = nil,
              actionTitle: String? = nil,
              action: (() -> Void)? = nil,
              dismissTitle: String? = nil,
              onDismiss: (() -> Void)? = nil) {
        hideWork?.cancel()
        current = Message(text: text, warning: warning,
                          actionTitle: actionTitle, action: action,
                          dismissTitle: dismissTitle, onDismiss: onDismiss)
        ToastPanelController.shared.present(
            text: text, warning: warning,
            actionTitle: actionTitle, action: action,
            dismissTitle: dismissTitle, onDismiss: onDismiss)
        let work = DispatchWorkItem { [weak self] in self?.dismiss() }
        hideWork = work
        DispatchQueue.main.asyncAfter(deadline: .now() + seconds, execute: work)
    }

    func dismiss() {
        hideWork?.cancel()
        hideWork = nil
        current = nil
        ToastPanelController.shared.hide()
    }
}
```

```swift
/// Borderless, non-activating panel that floats failure toasts above everything,
/// independent of the popover's lifetime (the old popover-only toast could not
/// reach a paste failure that happens after close).
@MainActor
final class ToastPanelController {
    static let shared = ToastPanelController()

    private var panel: NSPanel?
    private var keyMonitor: Any?

    func present(text: String,
                 warning: Bool,
                 actionTitle: String?,
                 action: (() -> Void)?,
                 dismissTitle: String?,
                 onDismiss: (() -> Void)?) {
        if panel == nil {
            let p = NSPanel(contentRect: .zero,
                            styleMask: [.borderless, .nonactivatingPanel],
                            backing: .buffered, defer: false)
            p.level = .statusBar
            p.isOpaque = false
            p.backgroundColor = .clear
            p.hasShadow = false
            p.isMovableByWindowBackground = false
            p.hidesOnDeactivate = false
            p.becomesKeyOnlyIfNeeded = true
            panel = p
            keyMonitor = NSEvent.addLocalMonitorForEvents(matching: .keyDown) { [weak self] evt in
                if evt.keyCode == 53 {          // Escape dismisses (S-003 access line)
                    MainActor.assumeIsolated {
                        ToastCenter.shared.dismiss()
                    }
                    return nil
                }
                return evt
            }
        }
        let toast = CopiedToast(text: text,
                                warning: warning,
                                actionTitle: actionTitle,
                                action: action,
                                dismissTitle: dismissTitle,
                                onDismiss: onDismiss)
        let hosting = NSHostingView(rootView: toast)
        panel?.contentView = hosting
        let size = hosting.fittingSize
        if let screen = NSScreen.main {
            let x = screen.frame.midX - size.width / 2
            let y = screen.frame.maxY - 72      // just under the menu bar
            panel?.setFrame(NSRect(origin: CGPoint(x: x, y: y), size: size), display: false)
        }
        panel?.orderFrontRegardless()
    }

    func hide() {
        panel?.orderOut(nil)
        if let keyMonitor {
            NSEvent.removeMonitor(keyMonitor)
            self.keyMonitor = nil
        }
    }
}
```

```swift
// Sources/KopieCore/Support/ShortcutConflict.swift
import Foundation

/// Pure conflict rules for paste bindings — the settings UI renders the
/// violations, this decides them (no silent override, S-004).
public enum ShortcutConflict {

    /// `candidate` (a full key + modifier combo) versus one saved role slot.
    public static func conflict(candidate: HotKeySpec,
                                against role: PasteBindings.Role,
                                bindings: PasteBindings) -> PasteBindings.Role? {
        let other: HotKeySpec
        switch role {
        case .globalHotkey: other = bindings.globalHotkey
        case .directPaste: other = bindings.directPaste
        case .plainModifier: return plainModifierConflict(plainModifier: bindings.plainModifier, combo: candidate) ? .plainModifier : nil
        }
        return other == candidate ? role : nil
    }

    /// `candidate` versus every other saved slot; first conflict wins.
    public static func conflict(candidate: HotKeySpec, others: [(role: String, spec: HotKeySpec?)]) -> (role: String, spec: HotKeySpec?)? {
        others.first { $0.spec == candidate }
    }

    /// A held plain modifier (no key) shadows any combo whose modifier mask is
    /// exactly that modifier — e.g. binding "Paste as plain" to ⌃ while a combo
    /// is also bare ⌃ would make one keystroke do both jobs.
    public static func plainModifierConflict(plainModifier: UInt32, combo: HotKeySpec) -> Bool {
        combo.modifiers == plainModifier
    }
}
```
